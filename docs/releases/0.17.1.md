## Portable verification

In a clone of the public repository containing the original preparation commit, save the
public archive outside the checkout and run this Python 3 snippet with its path as the
first argument. It does not install or execute the package, extract files to disk, repack,
or require the original account-holder packet.

```python
import base64, hashlib, json, subprocess, sys, tarfile
from pathlib import Path, PurePosixPath

source = "7d4de763e49024e1446805d98358b0503d1a90bc"
assert subprocess.check_output(["git", "rev-parse", source + "^{tree}"], text=True).strip() == "94def72fb32f01685e574469e4502252751213fd"
raw = Path(sys.argv[1]).read_bytes()
assert len(raw) == 483563
assert hashlib.sha256(raw).hexdigest() == "32c51fbbc39a8b51e631abeb92776c8642069e0844387413321eb95f7c3782f6"
assert hashlib.sha1(raw).hexdigest() == "e20f4e68d614cfbec7dfc1a07448eb3e07d95475"
assert "sha512-" + base64.b64encode(hashlib.sha512(raw).digest()).decode() == "sha512-eNqVR0KjClUSAXsGG8wP4rz+mkQ1r7OZ6m61GOajWorcKr4wm4VPP6DmxswZnqytKADY0n+xky5J8vEmR10LYA=="
files = {}
with tarfile.open(sys.argv[1]) as archive:
    for member in archive:
        path = PurePosixPath(member.name)
        assert member.isfile() and path.parts[0] == "package"
        assert not path.is_absolute() and ".." not in path.parts
        name = path.relative_to("package").as_posix()
        assert name not in files
        files[name] = archive.extractfile(member).read()
assert len(files) == 442
source_count = 0
for name, contents in files.items():
    if not name.startswith("dist/"):
        assert contents == subprocess.check_output(["git", "show", source + ":" + name]), name
        source_count += 1
assert source_count == 105
manifest = json.loads(files["package.json"])
assert manifest["name"] == "@ayayaq/vivi" and manifest["version"] == "0.17.1"
assert manifest["dependencies"] == {"yaml": "2.9.1"}
assert len(manifest["exports"]) == 17
def check_targets(value):
    if isinstance(value, str):
        assert value.removeprefix("./") in files, value
        return 1
    return sum(check_targets(child) for child in value.values())
assert check_targets(manifest["exports"]) == 68
print("Verified archive identity, 105 source files and 68 export targets")
```

## Scope and limits

The patch requests SSE explicitly and accepts validated bounded SSE framing when media
type metadata is missing or incorrect. It retains terminal-event, cancellation and size
checks, rejects partial output and adds bounded opt-in sanitized diagnostics. Dependencies,
authentication, endpoint selection, account selection and billing are unchanged.
No live credentials, sign-in or paid inference were used; the reported ChatGPT session's
actual response body and root cause remain unknown. CLI adoption, host UI behavior and
installed-host verification remain separately scoped.
